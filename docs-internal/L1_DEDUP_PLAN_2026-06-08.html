// Task 1-1 핵심 코드
function splitLegacyAddress(address) {
    if (!address) return { postcode: null, road: null };
    const m = address.match(/^\((\d{5})\)\s*(.+)$/);
    return m ? { postcode: m[1], road: m[2] } : { postcode: null, road: address };
}

// Task 1-4 핵심 구조
filterAndRenderLogs() {
    const filtered = this.sampleLogs.filter(log =>
        this.matchesNameFilter(log) && this.matchesReceptionFilter(log) &&
        this.matchesDateFilter(log) && this.matchesCompletedFilter(log) &&
        this.matchesTypeSpecificFilters(log));   // ← 훅 (soil: 필지+목적)
    this.renderLogs(filtered);
    this.updateSearchButtonState();
}
